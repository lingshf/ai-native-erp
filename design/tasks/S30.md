### S30 统一量法

三份 REFAC 任务书、开发 PR 与合并后复测都用同一脚本量函数长度与嵌套深度（D615）；任务书只引用本节，不另附脚本。仓库 `typescript` 为 7.0.2，不提供 `fnlen.cjs` 所需的 JS 编译器 API，故改用根 `package.json` 已声明的 `@babel/parser`。行数口径与 `fnlen.cjs` 相同：在 `4d296bf` 上，三文件 7 个目标函数的行数两者一致（200、189、131、182、124、100、106）。

- **行数**：数函数首行到末行之间的“代码行”。某行只要有一个非空白字符落在解析器报告的所有注释区间（`ast.comments`）之外，就是代码行。因此空行、纯 `//` 行、整行 `/* … */`、多行注释中间的行都不计；注释与代码同在一行（如 `/* c */ a();`、多行注释收尾的 `*/ b();`）计为代码行。内联回调、JSX 的行，以及具名内层函数的行，都计入所在的外层函数。
- **嵌套**：函数体顶层记 0 层。以下各加 1 层：`if` 的 then / else 分支、循环体、`switch` 的 `case`、`try` / `catch` / `finally` 块，以及任何不具名函数（调用实参、JSX 属性回调、立即调用的函数）的函数体。受控语句体不带花括号时同样计层；`else if` 与前面的 `if` 同层；`?:`、`&&` 与 JSX 元素层级不计层。
- **单独计量的函数**：函数声明、对象方法、赋给变量或属性的函数，各自单独输出一行，有自己的行数与层数。这类函数的**层数**与外层互相独立：外层计层时不进入其函数体。它的**行数**仍计入外层：外层行数按首行到末行的代码行计，不扣除内层函数。
- **运行**：把脚本存为仓库外的 `s30-measure.cjs`，在仓库根目录运行 `node <保存位置>\s30-measure.cjs <修订> <文件…>`，脚本用 `git show <修订>:<文件>` 读入。输出每行依次为行数、最深层数、`文件:首行-末行`、函数名，以制表符分隔。

```js
const { parse } = require(require.resolve("@babel/parser", { paths: [process.cwd()] }));
const { execFileSync } = require("node:child_process");
const [rev, ...files] = process.argv.slice(2);
const FN = new Set(["FunctionDeclaration", "FunctionExpression", "ArrowFunctionExpression", "ObjectMethod", "ClassMethod"]);
const LOOP = new Set(["ForStatement", "ForInStatement", "ForOfStatement", "WhileStatement", "DoWhileStatement"]);
const NAMED_PARENT = new Set(["VariableDeclarator", "ObjectProperty", "ClassProperty"]);
const isCallback = (n, parent) => FN.has(n.type) && !n.id && !n.key && !NAMED_PARENT.has(parent?.type);
const children = (n) =>
  Object.entries(n).flatMap(([k, v]) =>
    ["loc", "leadingComments", "trailingComments", "innerComments", "extra"].includes(k) ? [] :
    Array.isArray(v) ? v.filter((c) => c?.type) : v?.type ? [v] : []);
// A line counts as code if it has a non-whitespace character outside every comment range reported by the parser.
function codeLineSet(text, comments) {
  const inComment = new Uint8Array(text.length);
  for (const c of comments) inComment.fill(1, c.start, c.end);
  const lines = new Set();
  let line = 1;
  for (let i = 0; i < text.length; i++) {
    if (text[i] === "\n") line++;
    else if (!inComment[i] && !/\s/.test(text[i])) lines.add(line);
  }
  return lines;
}
// Roadmap rule 2: if/else, loop, switch, try/catch bodies and inline callback bodies add one level,
// whether or not the body has braces; else-if stays at the level of its if; named inner functions are measured separately.
function depth(fn) {
  let max = 0;
  const enter = (s, d, parent) => { if (s) { max = Math.max(max, d); visit(s, d, parent); } };
  function visit(n, d, parent) {
    if (n !== fn && FN.has(n.type)) { if (isCallback(n, parent)) enter(n.body, d + 1, n); return; }
    if (n.type === "IfStatement") {
      visit(n.test, d, n); enter(n.consequent, d + 1, n);
      if (n.alternate?.type === "IfStatement") visit(n.alternate, d, n); else enter(n.alternate, d + 1, n);
      return;
    }
    if (LOOP.has(n.type)) { for (const c of children(n)) c === n.body ? enter(c, d + 1, n) : visit(c, d, n); return; }
    if (n.type === "TryStatement") { enter(n.block, d + 1, n); enter(n.handler?.body, d + 1, n); enter(n.finalizer, d + 1, n); return; }
    if (n.type === "SwitchStatement") { visit(n.discriminant, d, n); for (const c of n.cases) enter(c, d + 1, n); return; }
    for (const c of children(n)) visit(c, d, n);
  }
  if (fn.body.type === "BlockStatement") for (const s of fn.body.body) visit(s, 0, fn.body); else visit(fn.body, 0, fn);
  return max;
}
for (const file of files) {
  const text = execFileSync("git", ["show", `${rev}:${file}`], { encoding: "utf8" });
  const ast = parse(text, { sourceType: "module", plugins: ["typescript", ...(file.endsWith("x") ? ["jsx"] : [])] });
  const code = codeLineSet(text, ast.comments);
  const codeLines = (n) => { let k = 0; for (let l = n.loc.start.line; l <= n.loc.end.line; l++) if (code.has(l)) k += 1; return k; };
  (function walk(n, parent) {
    if (FN.has(n.type) && !isCallback(n, parent) && n.body) {
      const name = n.id?.name ?? n.key?.name ?? parent?.id?.name ?? parent?.key?.name ?? "<anonymous>";
      console.log([codeLines(n), depth(n), `${file}:${n.loc.start.line}-${n.loc.end.line}`, name].join("\t"));
    }
    for (const c of children(n)) walk(c, n);
  })(ast.program, null);
}
```
